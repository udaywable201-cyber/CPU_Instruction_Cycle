03 / PROGRAMMING LANGUAGE

Python example · simulator.py
# CPU Instruction Simulator

def execute_instruction(instruction, registers, memory):
    # Convert instruction to uppercase and split into parts
    parts = instruction.upper().replace(",", " ").split()

    # Check if instruction is empty
    if len(parts) == 0:
        print("Please enter an instruction")
        return

    # Get the instruction name
    op = parts[0]

    # Perform arithmetic operations
    if op in ["ADD", "SUB", "MUL", "DIV"]:
        a = float(parts[1])
        b = float(parts[2])

        if op == "ADD":
            result = a + b       # Addition
        elif op == "SUB":
            result = a - b       # Subtraction
        elif op == "MUL":
            result = a * b       # Multiplication
        elif op == "DIV":
            if b == 0:
                print("Cannot divide by zero")
                return
            result = a / b       # Division

        registers["ACC"] = result  # Store result in ACC
        print("Result:", result)

    # Load a number into the accumulator
    elif op == "LOAD":
        registers["ACC"] = float(parts[1])
        print("Loaded value:", registers["ACC"])

    # Store ACC value in memory location 0
    elif op == "STORE":
        memory[0] = registers["ACC"]
        print("Stored value:", memory[0])

    else:
        print("Invalid instruction")


# Initialize CPU register and memory
registers = {"ACC": 0}
memory = [0] * 8

# Take an instruction from the user
instruction = input("Enter instruction: ")

# Execute the instruction
execute_instruction(instruction, registers, memory)

# Display final register and memory values
print("ACC Register:", registers["ACC"])
print("Memory:", memory)

04 / TESTING AND RESULTS

Testing procedure

  1. Open this HTML file in a modern browser.
  2. Enter an instruction or choose one from the dropdown.
  3. Click Run Cycle to complete the cycle.
  4. Click Next Step repeatedly to inspect each stage.
  5. Verify register values, memory, result and error messages.
  6. Use Reset CPU or Reset All to check reset behavior.
